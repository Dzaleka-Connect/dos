# Content usage preferences for search and AI systems.
#
# This site exists so that people can find accurate, current information about
# Dzaleka Refugee Camp, including through assistants and agents. Our position:
#
#   search=yes     Index our pages and link to them.
#   ai-input=yes   Use our content to answer questions about Dzaleka, provided
#                  you attribute it and link to the source page. This is what
#                  our public API, /llms.txt and MCP server are published for.
#   ai-train=no    Do not use our content to train or fine-tune models. This is
#                  a community's own record of itself; read it and cite it,
#                  do not absorb it.

# artworks carry separate rights - check the credit on each item before reuse.
